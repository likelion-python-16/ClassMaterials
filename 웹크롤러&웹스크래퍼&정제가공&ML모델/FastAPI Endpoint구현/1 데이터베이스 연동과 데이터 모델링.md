src/core/database.py
```python
import os
from dotenv import load_dotenv
from sqlalchemy import URL, create_engine
from sqlalchemy.orm import sessionmaker, declarative_base


# 환경변수 로드
load_dotenv(verbose=True)

# SQLAlchemy 접속 URL 구성
DATABASE_URL = os.getenv("DATABASE_URL")
if not DATABASE_URL:
    DATABASE_URL = URL.create(
        drivername=os.getenv("DATABASE_DRIVER"),   
        username=os.getenv("DATABASE_USERNAME"),
        password=os.getenv("DATABASE_PASSWORD"),
        host=os.getenv("DATABASE_HOST"),
        port=os.getenv("DATABASE_PORT"),
        database=os.getenv("DATABASE_NAME"),
        # MySQL이면 charset 지정(emoji 등 포함)
        query={"charset": "utf8mb4"} if "mysql" in (os.getenv("DATABASE_DRIVER") or "") else {},
    )

# DB 엔진 생성
engine = create_engine(
    DATABASE_URL,
    pool_pre_ping=True,  # 죽은 커넥션 자동 감지
    pool_recycle=1800,   # MySQL wait_timeout 회피
    echo=os.getenv("SQL_ECHO", "false").lower() == "true",  
    # 필요 시 SQL 로그
    future=True,
)

# 세션 로컬 설정
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False, future=True)
# - SessionLocal()을 실행하면 DB 세션이 만들어집니다.   
# - 일반적으로 FastAPI에서는 이걸 의존성 주입으로 사용합니다.


# Base 클래스 (모든 모델의 부모가 됨)
Base = declarative_base()
# 모든 SQLAlchemy 모델 클래스는 이 Base를 상속받아야 합니다.

# 5) FastAPI 의존성
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
```

src/models/restaurant_model.py
```python
from sqlalchemy import (
    Column, BigInteger, Integer, String, Text, Boolean, Time,
    DECIMAL, ForeignKey, Table, Index, CheckConstraint
)
from sqlalchemy.orm import relationship
from src.core.database import Base

# ManyToMany 중간 테이블 이름/컬럼은 실제 DB에서 확인해 맞추세요.
# 기본 규칙: <app>_<model>_<field> → restaurant_restaurant_tags (through 모델이 있으면 그 이름)
restaurant_tag = Table(
    "restaurant_restaurant_tags",  # 실제 이름 확인 필수
    Base.metadata,
    Column("id", BigInteger, primary_key=True, autoincrement=True),  # Django through 모델이면 id 존재
    Column("restaurant_id", BigInteger, ForeignKey("restaurant_restaurant.id", ondelete="CASCADE"), nullable=False),
    Column("tag_id", BigInteger, ForeignKey("restaurant_tag.id", ondelete="CASCADE"), nullable=False),
)

class Restaurant(Base):
    __tablename__ = "restaurant_restaurant"

    id = Column(BigInteger, primary_key=True, autoincrement=True)

    name = Column(String(100), nullable=False, index=True)
    branch_name = Column(String(100), nullable=True, index=True)
    description = Column(Text, nullable=True)
    address = Column(String(255), nullable=False, index=True)
    feature = Column(String(255), nullable=False)
    is_closed = Column(Boolean, nullable=False, default=False)

    latitude  = Column(DECIMAL(16, 12), nullable=False, default="0.0000", index=True)
    longitude = Column(DECIMAL(16, 12), nullable=False, default="0.0000", index=True)

    phone = Column(String(16), nullable=True)
    rating = Column(DECIMAL(3, 2), nullable=False, default="0.0")
    rating_count = Column(Integer, nullable=False, default=0)

    start_time = Column(Time, nullable=True)
    end_time = Column(Time, nullable=True)
    last_order_time = Column(Time, nullable=True)

    category_id = Column(BigInteger, ForeignKey("restaurant_restaurantcategory.id", ondelete="SET NULL"), nullable=True)
    region_id   = Column(BigInteger, ForeignKey("restaurant_region.id",           ondelete="SET NULL"), nullable=True)

    category = relationship("RestaurantCategory", back_populates="restaurants")
    region   = relationship("Region", back_populates="restaurants")

    tags = relationship("Tag", secondary=restaurant_tag, back_populates="restaurants")

    __table_args__ = (
        CheckConstraint("rating_count >= 0", name="ck_rating_count_nonneg"),
        Index("ix_restaurant_lat_lng", "latitude", "longitude"),
    )

class RestaurantCategory(Base):
    __tablename__ = "restaurant_restaurantcategory"
    id = Column(BigInteger, primary_key=True, autoincrement=True)
    name = Column(String(100), nullable=False)
    cuisine_type_id = Column(BigInteger, ForeignKey("restaurant_cuisinetype.id"), nullable=True)  
    restaurants = relationship("Restaurant", back_populates="category")

class Region(Base):
    __tablename__ = "restaurant_region"
    id = Column(BigInteger, primary_key=True, autoincrement=True)
    sido = Column(String(100), nullable=False)
    sigungu = Column(String(100), nullable=True)
    eupmyendong = Column(String(100), nullable=True)
    restaurants = relationship("Restaurant", back_populates="region")

class Tag(Base):
    __tablename__ = "restaurant_tag"
    id = Column(BigInteger, primary_key=True, autoincrement=True)
    name = Column(String(100), nullable=False)
    restaurants = relationship("Restaurant", secondary=restaurant_tag, back_populates="tags")
```

`Restaurant`
- Django 모델: `restaurant/models.py` 안의 `Restaurant` 모델과 1:1 매핑해야 함.
    - 필드: `name`, `branch_name`, `description`, `address`, `feature`, `is_closed`, `latitude`, `longitude`, `phone`, `rating`, `rating_count`, `start_time`, `end_time`, `last_order_time`, `category`, `region` 
    - PK: `id` (Django 기본 `BigAutoField`이면 FastAPI도 `BigInteger`)
    - FK: `category_id` → `RestaurantCategory`, `region_id` → `Region`
    - ManyToMany: `tags` → 중간테이블 `restaurant_restaurant_tags` 사용.
---
`RestaurantCategory`
- Django 모델: `RestaurantCategory` 모델과 매핑.
    - 필드: `name` 동일, FK `cuisine_type_id`는 Django `RestaurantCategory`에서 `cuisine_type` (ForeignKey)와 동일.
    - FK 대상: `CuisineType` (`restaurant_cuisinetype` 테이블)
---
`Region`
- Django 모델: `Region` 모델과 매핑.
    - 다이어그램 상 `sido`, `sigungu`, `eupmyendong` 컬럼이 있어서 그대로 매핑됨.
    - `restaurants` 관계는 `Restaurant.region`의 역참조.
---
`Tag` & `restaurant_tag` (M2M 테이블)
- Django 모델: `Tag` 모델과 매핑 + `Restaurant.tags` 필드의 M2M 관계.
    - `restaurant_tag` 테이블 = Django `Tag` 모델과 매핑.
    - 중간테이블 `restaurant_restaurant_tags` = Django가 자동 생성한 through 테이블(`Restaurant.tags`).
---
고객 전용 여부
`Restaurant`, `RestaurantCategory`, `Region`, `Tag` 전부 Django의 관리자(Admin) 페이지에서 직접 관리하는 공유 테이블입니다.
- 관리자 전용 데이터베이스에 속함.
- FastAPI는 이걸 그대로 읽거나(조회 API) 쓰는(등록/수정 API) 용도로 사용.
---
연결점 (FastAPI ↔ Django)
- DB 스키마 레벨에서 직접 연결
    - 둘 다 같은 DB를 바라보고, 테이블명을 동일하게(`restaurant_restaurant` 등) 지정.
- ORM 계층에서 따로 연결 로직은 없음
    - FastAPI는 SQLAlchemy ORM, Django는 Django ORM이 같은 테이블을 바라보게끔 설정.
- 변환은 Pydantic 스키마에서 처리
    - 고객에게 보여줄 필드만 Pydantic 모델로 가공해서 응답.
---
주의할 점
마이그레이션 주도권
    - 공유 테이블은 Django가 마이그레이션 주도.
    - FastAPI에서는 절대 `Base.metadata.create_all()`로 생성하거나 변경하면 안 됨.

from sqlalchemy import (
    Column, BigInteger, String, ForeignKey, Table, UniqueConstraint, DateTime
)
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func
from src.core.database import Base

# ------------------------------
# 1. Django 공유 테이블
# ------------------------------
class CuisineType(Base):
    """
    Django restaurant_cuisinetype 매핑
    """
    __tablename__ = "restaurant_cuisinetype"  
    # Django: <app>_<model>

    id = Column(BigInteger, primary_key=True, autoincrement=True)
    name = Column(String(20), nullable=False)  
    # Django: CharField(max_length=20)

# ------------------------------
# 2. 고객전용 확장 스키마 (fdc_*)
# ------------------------------
class CuisineTypeCategory(Base):
    """
    고객전용: 음식 종류 카테고리
    Django에는 없음. FastAPI 서비스 로직 전용.
    """
    __tablename__ = "fdc_cuisine_type_categories"

    id = Column(BigInteger, primary_key=True, autoincrement=True)
    name = Column(String(100), nullable=False, unique=True, index=True)
    created_at = Column(DateTime, nullable=False, server_default=func.now())
    updated_at = Column(DateTime, nullable=False, server_default=func.now(), onupdate=func.now())

    cuisine_types = relationship("FdcCuisineType", back_populates="cuisine_type_category")

class FdcCuisineType(Base):
    """
    고객전용: Django CuisineType과는 별개의 테이블.
    추천 서비스 등에서 자유롭게 사용.
    """
    __tablename__ = "fdc_cuisine_types"

    id = Column(BigInteger, primary_key=True, autoincrement=True)
    name = Column(String(100), nullable=False)

    cuisine_type_category_id = Column(
        BigInteger,
        ForeignKey("fdc_cuisine_type_categories.id", ondelete="SET NULL"),
        nullable=True
    )
    created_at = Column(DateTime, nullable=False, server_default=func.now())
    updated_at = Column(DateTime, nullable=False, server_default=func.now(), onupdate=func.now())

    cuisine_type_category = relationship("CuisineTypeCategory", back_populates="cuisine_types")

# ------------------------------
# 3. 고객전용: 식당과 fdc_cuisine_types 연결
# ------------------------------
fdc_restaurant_cuisine_type = Table(
    "fdc_restaurant_cuisine_type",
    Base.metadata,
    Column("restaurant_id", BigInteger,
           ForeignKey("restaurant_restaurant.id", ondelete="CASCADE"),
           primary_key=True),
    Column("cuisine_type_id", BigInteger,
           ForeignKey("fdc_cuisine_types.id", ondelete="CASCADE"),
           primary_key=True),
    UniqueConstraint("restaurant_id", "cuisine_type_id", name="uq_fdc_restaurant_cuisine_type"),

SQLlite로 fast api에서 모델코드 생성후 실제 테이블을 생성하는 코드
```python
import os
from dotenv import load_dotenv

from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from sqlalchemy.engine.url import URL

# .env 파일 로드
load_dotenv(verbose=True)

# 환경변수 읽기
driver = os.getenv("DATABASE_DRIVER", "sqlite")
db_name = os.getenv("DATABASE_NAME", "fastdining.db")

# SQLAlchemy 접속 URL 구성
if driver == "sqlite":
    # SQLite의 경우 로컬 파일을 사용
    SQLALCHEMY_DATABASE_URL = f"sqlite:///./{db_name}"
    connect_args = {"check_same_thread": False}
else:
    SQLALCHEMY_DATABASE_URL = URL.create(
        drivername=driver,
        username=os.getenv("DATABASE_USERNAME"),
        password=os.getenv("DATABASE_PASSWORD"),
        host=os.getenv("DATABASE_HOST"),
        port=os.getenv("DATABASE_PORT"),
        database=db_name,
    )
    connect_args = {}

# DB 엔진 생성
engine = create_engine(SQLALCHEMY_DATABASE_URL, connect_args=connect_args)

# 세션 로컬 설정
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Base 클래스 (모든 모델들이 상속받음)
Base = declarative_base()


# 선택: 테이블 자동 생성 함수 (필요 시 실행)
def init_db():
    from src.models import article_model
    # 여러 모델을 사용할 경우 여기에 추가로 import
    Base.metadata.create_all(bind=engine)
```
