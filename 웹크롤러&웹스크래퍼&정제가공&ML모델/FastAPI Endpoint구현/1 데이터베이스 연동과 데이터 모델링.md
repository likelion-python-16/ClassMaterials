src/core/database.py
```python
import os
from dotenv import load_dotenv

from sqlalchemy import URL, create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
'''
- os : 환경 변수 읽기용  
- load_dotenv(): .env 파일에 정의된 DB 정보(비밀번호등)를 메모리에 로드 
- URL.create(): SQLAlchemy에서 DB 주소를 객체로 생성해줌 (MySQL, PostgreSQL 등 호환)   
- create_engine(): 실제 DB 연결 객체 생성   
- sessionmaker(): DB에 쿼리할 때 사용할 세션 객체 생성기  
- declarative_base(): SQLAlchemy 모델들이 상속받는 기본 클래스
'''

# 환경변수 로드
load_dotenv(verbose=True)

# SQLAlchemy 접속 URL 구성
SQLALCHEMY_DATABASE_URL = URL.create(
    drivername=os.getenv("DATABASE_DRIVER"), # 예: "pymysql"
    username=os.getenv("DATABASE_USERNAME"), # 예: "root"
    password=os.getenv("DATABASE_PASSWORD"), # 예: "1234"
    host=os.getenv("DATABASE_HOST"),         # 예: "localhost"
    port=os.getenv("DATABASE_PORT"),         # 예: "3306"
    database=os.getenv("DATABASE_NAME")      # 예: "fast_db"
)

# DB 엔진 생성
engine = create_engine(SQLALCHEMY_DATABASE_URL)
# - 실제로 DB와 연결되는 엔진 객체입니다.   
# - 이 엔진을 통해 SQL을 실행하거나 테이블을 생성할 수 있어요.


# 세션 로컬 설정
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
# - SessionLocal()을 실행하면 DB 세션이 만들어집니다.   
# - 일반적으로 FastAPI에서는 이걸 의존성 주입으로 사용합니다.


# Base 클래스 (모든 모델의 부모가 됨)
Base = declarative_base()
# 모든 SQLAlchemy 모델 클래스는 이 Base를 상속받아야 합니다.
```

from sqlalchemy import Boolean, Column, ForeignKey, Integer, String, DateTime
from sqlalchemy.orm import relationship

from src.core.database import Base

from src.models import user_like_model, user_follow_model

class User(Base):
    __tablename__ = "auth_user"

    id = Column(Integer, primary_key=True, autoincrement=True)
    username = Column(String, unique=True, index=True, nullable=False)
    password = Column(String, nullable=False)

    first_name = Column(String, nullable=False)
    last_name = Column(String, nullable=False)

    email = Column(String, nullable=False)

    nickname = Column(String, nullable=True)
    bio = Column(String, nullable=True)

    is_superuser = Column(Boolean, nullable=False)
    is_staff = Column(Boolean, nullable=False)
    is_active = Column(Boolean, nullable=False)

    last_login = Column(DateTime, nullable=True)
    date_joined = Column(DateTime, nullable=False)

    access_token = relationship("AccessToken", back_populates="user")

SQLlite로 fast api에서 모델코드 생성후 실제 테이블을 생성하는 코드
```python
import os
from dotenv import load_dotenv

from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from sqlalchemy.engine.url import URL

# .env 파일 로드
load_dotenv(verbose=True)

# 환경변수 읽기
driver = os.getenv("DATABASE_DRIVER", "sqlite")
db_name = os.getenv("DATABASE_NAME", "fastdining.db")

# SQLAlchemy 접속 URL 구성
if driver == "sqlite":
    # SQLite의 경우 로컬 파일을 사용
    SQLALCHEMY_DATABASE_URL = f"sqlite:///./{db_name}"
    connect_args = {"check_same_thread": False}
else:
    SQLALCHEMY_DATABASE_URL = URL.create(
        drivername=driver,
        username=os.getenv("DATABASE_USERNAME"),
        password=os.getenv("DATABASE_PASSWORD"),
        host=os.getenv("DATABASE_HOST"),
        port=os.getenv("DATABASE_PORT"),
        database=db_name,
    )
    connect_args = {}

# DB 엔진 생성
engine = create_engine(SQLALCHEMY_DATABASE_URL, connect_args=connect_args)

# 세션 로컬 설정
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Base 클래스 (모든 모델들이 상속받음)
Base = declarative_base()


# 선택: 테이블 자동 생성 함수 (필요 시 실행)
def init_db():
    from src.models import article_model
    # 여러 모델을 사용할 경우 여기에 추가로 import
    Base.metadata.create_all(bind=engine)
```
