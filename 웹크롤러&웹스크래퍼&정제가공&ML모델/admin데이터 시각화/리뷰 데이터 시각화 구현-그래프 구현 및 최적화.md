 어드민의 특정 모델(AnalyticsDashboard)의 목록 화면을 대시보드로 사용.
- 목록 템플릿을 커스텀(`change_list_template`)하고, `changelist_view()`에서 차트용 데이터를 만들어 템플릿에 넘김.
- 템플릿은 `Chart.js`로 차트를 그림.

(1) `restaurant/admin.py` 대시보드용 Admin 등록
```python
@admin.register(AnalyticsDashboard)
class AnalyticsDashboardAdmin(admin.ModelAdmin):
    change_list_template = "admin/analytics_changelist.html"  
    # 커스텀 목록 템플릿
    
    list_per_page = 20 # 페이지네이션
    
    def has_add_permission(self, request): return False
    def has_change_permission(self, request, obj=None): return False
    def has_delete_permission(self, request, obj=None): return False

    @method_decorator(cache_page(60*10))
    def changelist_view(self, request, extra_context=None):
        
        # GET 파라미터(start, end)로 기간 필터
        start = request.GET.get("start")
        end   = request.GET.get("end")

        def rng(qs, field="created_at"):
            if start: qs = qs.filter(**{f"{field}__date__gte": start})
            if end:   qs = qs.filter(**{f"{field}__date__lte": end})
            return qs

        # 1) 월별 리뷰 수 (view → review로 교체)
        reviews_monthly_base_qs = FdcReviews.objects.using("fdc")
        reviews_monthly_qs = rng(reviews_monthly_base_qs, field="created_at")
        monthly_reviews = (
            reviews_monthly_qs
            .annotate(month=TruncMonth("created_at"))
            .values("month")
            .annotate(total=Count("id"))
            .order_by("month")
        )
        mv_labels = [x["month"].strftime("%Y-%m") for x in monthly_reviews]
        mv_data = [x["total"] or 0 for x in monthly_reviews]

        # 2) 상위 카테고리
        top_cuisines_qs = FdcRestaurantHasCuisineTypes.objects.using("fdc")
        top_cuisines = (top_cuisines_qs.values("cuisine_type__name")
            .annotate(n=Count("id")).order_by("-n")[:10])
        tc_labels = [(x["cuisine_type__name"] or "미상") for x in top_cuisines]
        tc_data   = [x["n"] for x in top_cuisines]

        # 3) 평점 분포
        rating_qs = rng(FdcReviews.objects.using("fdc"))
        rating_dist = (rating_qs.exclude(rating__isnull=True)
            .values("rating").annotate(n=Count("id")).order_by("rating"))
        rd_labels = [str(x["rating"]) for x in rating_dist]
        rd_data   = [x["n"] for x in rating_dist]

        # 4) 태그 TOP 15
        tags_qs = FdcRestaurantHasTags.objects.using("fdc")
        top_tags = (tags_qs.values("tag__name")
            .annotate(n=Count("id")).order_by("-n")[:15])
        tt_labels = [(x["tag__name"] or "미상") for x in top_tags]
        tt_data   = [x["n"] for x in top_tags]

        charts = {
            "monthly_views": {"labels": mv_labels, "data": mv_data},
            "top_cuisines":  {"labels": tc_labels, "data": tc_data},
            "rating_dist":   {"labels": rd_labels, "data": rd_data},
            "top_tags":      {"labels": tt_labels, "data": tt_data},
        }
        extra = {
            "title": "운영 대시보드",
            "charts_json": mark_safe(json.dumps(charts, ensure_ascii=False)),
            "date_range": {"start": start or "", "end": end or ""},
        }
# 기본 changelist 컨텍스트(cl/opts 등)를 유지해야 하므로 super로 렌더
        return super().changelist_view(request, extra_context={**(extra_context or {}), **extra})
```

(2) 읽기전용 M2M 회피(마이그레이션 없이 보기만)
`RestaurantAdmin`에서 `tags` 필드를 폼/필터에서 제거하고, `fdc`에서 조회해 보여주는 `tag_list`로 대체: 스키마 건드리지 않고 보기전용 코드
```python
@admin.register(Restaurant)
class RestaurantAdmin(admin.ModelAdmin):
    list_display = ("id","name","branch_name","is_closed","phone","rating","rating_count","region")
    fields = ("name","branch_name","category","is_closed","phone","latitude","longitude","tag_list")
    readonly_fields = ("rating","rating_count","region","tag_list")
    exclude = ("tags",)
    list_per_page = 20
    inlines = (RestaurantMenuInline, RestaurantImageInline)

    def tag_list(self, obj):
        if not obj or not obj.pk: return "-"
        with connections["fdc"].cursor() as cur:
            cur.execute("""
                SELECT t.name
                FROM fdc_restaurant_has_tags ht
                JOIN fdc_tags t ON t.id = ht.tag_id
                WHERE ht.restaurant_id = %s
                ORDER BY t.name
            """, [obj.pk])
            rows = cur.fetchall()
        return ", ".join(r[0] for r in rows) if rows else "-"
    tag_list.short_description = "태그(읽기전용)"
```

Chart.js [https://www.chartjs.org/docs/latest/]

챠트의 디자인을 변경하려면
- 공통 옵션(COMMON_OPTS)
    - 축, 그리드, 범례, 툴팁, 반응형 설정 등 차트 전반의 레이아웃·UI 요소 조정
    - 여기서 바꾸면 모든 차트에 동일하게 적용
    - 예: y축 0부터 시작, 툴팁 포맷, 범례 위치, 그리드 색상 등
        
- 차트 데이터셋(datasets)
    - 개별 차트의 선/막대 색상, 두께, 점 모양, 채우기, segment 스타일 조정
    - `mkLineChart` / `mkBarChart` 함수 안의 `datasets: [{ ... }]` 부분
    - 예: `borderColor`, `backgroundColor`, `pointRadius`, `tension`, `barThickness` 등

- 차트 레이아웃·UI 스타일 → `COMMON_OPTS`
- 그래프 선·막대 자체 스타일 → `datasets` 내부 옵션
    
이 두 군데만 이해하면 Chart.js 샘플의 디자인 코드를 내 프로젝트에 그대로 적용할 수 있습니다.
